#include <WiFi.h>
#include <PubSubClient.h>

// ===== Credenciais WiFi =====
const char* ssid = "VALENTINI FIBRA";
const char* password = "08082005G";

// ===== Configuracoes do Broker MQTT =====
const char* mqtt_server = "broker.hivemq.com"; // ex: "broker.hivemq.com" ou IP do seu broker
const int mqtt_port = 1883;
const char* mqtt_user = "";     // deixe vazio se o broker nao exigir autenticacao
const char* mqtt_password = ""; // deixe vazio se o broker nao exigir autenticacao
const char* mqtt_topic = "sensores/dados";

#define TIPO_UMIDADE 0
#define TIPO_TEMPERATURA 1

// Structure example to send data
// Deve ser igual a struct usada no lado que recebe (subscriber)
typedef struct struct_message {
  uint8_t tipo;
  float dado;
} struct_message;

WiFiClient espClient;
PubSubClient client(espClient);

void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Conectando a ");
  Serial.println(ssid);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("");
  Serial.println("WiFi conectado");
  Serial.print("Endereco IP: ");
  Serial.println(WiFi.localIP());
}

void reconnect() {
  // Loop ate reconectar ao broker MQTT
  while (!client.connected()) {
    Serial.print("Tentando conexao MQTT...");
    String clientId = "ESP32Client-";
    clientId += String(random(0xffff), HEX);

    if (client.connect(clientId.c_str(), mqtt_user, mqtt_password)) {
      Serial.println("conectado");
    } else {
      Serial.print("falhou, rc=");
      Serial.print(client.state());
      Serial.println(" tentando novamente em 5 segundos");
      delay(5000);
    }
  }
}

void setup() {
  // Init Serial Monitor
  Serial.begin(115200);

  setup_wifi();

  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  struct_message msg = {
    .tipo = TIPO_UMIDADE,
    .dado = ((float)random(0, 10000) / 10000.0) * 100.0,
  };

  // Envia dados mockados via MQTT (payload binario com a mesma struct de antes)
  boolean result = client.publish(mqtt_topic, (uint8_t*)&msg, sizeof(msg));

  Serial.println(msg.dado);

  if (result) {
    Serial.println("Enviado com sucesso");
  } else {
    Serial.println("Erro ao enviar os dados");
  }

  delay(2000);
}
